model=pathlib.Path('/tmp/replika-asr-candidates/giga-v3');r=sherpa.OfflineRecognizer.from_transducer(encoder=str(model/'encoder.int8.onnx'),decoder=str(model/'decoder.onnx'),joiner=str(model/'joiner.onnx'),tokens=str(model/'tokens.txt'),num_threads=2,model_type='nemo_transducer',decoding_method='greedy_search',max_active_paths=4)

def context(spans,n,post):
 duration=n//16;ret=[]
 for start,end in spans:
  s=max(0,min(duration,start*1000//16000)-100);e=min(duration,end*1000//16000+post)
  if e>s:
   if ret and s<=ret[-1][1]:ret[-1][1]=max(ret[-1][1],e)
   else:ret.append([s,e])
 return ret
def transcribe(x,g,spans,post):
 texts=[]
 for start,end in context(spans,len(x),post):
  for begin in range(start,end,20000):
   stream=r.create_stream();stream.accept_waveform(16000,x[begin*16:min(end,begin+20000)*16]*g);r.decode_stream(stream);texts.append(stream.result.text)
 return ' '.join(texts).strip()

  ref=b.words(f['reference']);hyp=b.words(text)

neg={(x['id'],x['mode']):x for x in candidate['controls']}

 x=x.astype(np.float32)
