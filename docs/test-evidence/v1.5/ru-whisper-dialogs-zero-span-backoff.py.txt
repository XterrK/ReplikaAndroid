def save():(root/'zero-span-backoff.json').write_text(json.dumps(out,ensure_ascii=False,indent=2))
def vad(x,g,threshold):
 c=sherpa.VadModelConfig();c.silero_vad.model=str(repo/'model-pack/vad/silero_vad.onnx');c.silero_vad.threshold=threshold;c.silero_vad.min_silence_duration=.25;c.silero_vad.min_speech_duration=.25;c.silero_vad.max_speech_duration=20;c.sample_rate=16000;c.num_threads=1
 d=sherpa.VoiceActivityDetector(c,buffer_size_in_seconds=60);spans=[]
 def drain():
  while not d.empty():
   s=d.front;spans.append([s.start,s.start+len(s.samples)]);d.pop()
 for pos in range(0,len(x),512):
  block=np.zeros(512,dtype=np.float32);n=min(512,len(x)-pos);block[:n]=x[pos:pos+n]*g;d.accept_waveform(block);drain()
 d.flush();drain();return spans
def context(spans,n):
 duration=n//16;out=[]
 for start,end in spans:
  s=max(0,min(duration,start*1000//16000)-100);e=min(duration,end*1000//16000)
  if e>s:
   if out and s<=out[-1][1]:out[-1][1]=max(out[-1][1],e)
   else:out.append([s,e])
 return out
def policy(x,g):
 peak=float(np.max(np.abs(x)));first=vad(x,g,.5);retry=None
 if not first and peak>1e-6:retry=vad(x,g,.35)
 selected=first if first else (retry or [])
 if first:assert selected==first
 return first,retry,selected,peak

r=sherpa.OfflineRecognizer.from_transducer(encoder=str(model/'encoder.int8.onnx'),decoder=str(model/'decoder.onnx'),joiner=str(model/'joiner.onnx'),tokens=str(model/'tokens.txt'),num_threads=2,model_type='nemo_transducer',decoding_method='greedy_search',max_active_paths=4)
out['asrLoadSec']=time.perf_counter()-t

def transcribe(x,g,spans):
 texts=[];calls=0
 for start,end in context(spans,len(x)):
  for begin in range(start,end,20000):
   stream=r.create_stream();stream.accept_waveform(16000,x[begin*16:min(end,begin+20000)*16]*g);r.decode_stream(stream);texts.append(stream.result.text);calls+=1
 return ' '.join(texts).strip(),calls

 item={'id':f['id'],'peak':peak,'gain':g,'defaultSpans':first,'retrySpans':retry,'selectedSpans':selected,'contextMs':context(selected,len(x)),'firstPassPreserved':bool(first),'defaultHypothesis':base['hypothesis'],'hypothesis':text,'reference':f['reference'],'wordErrors':b.distance(ref,h),'referenceWords':len(ref),'charErrors':b.distance(list(' '.join(ref)),list(' '.join(h))),'referenceChars':len(' '.join(ref)),'asrCalls':calls}
 out['speech'].append(item);save();print('speech',f['id'],'retry',retry is not None,'spans',len(selected),'errors',item['wordErrors'],flush=True)
rng=np.random.default_rng(104);n=16000*15;t=np.arange(n)/16000
white=rng.normal(0,1,n);freq=np.fft.rfftfreq(n);pink=np.fft.irfft(np.fft.rfft(white)/np.sqrt(np.maximum(freq,1/n)),n);pink=(pink/np.max(np.abs(pink))*.004).astype(np.float32)
impulses=np.zeros(n,dtype=np.float32)
for start in range(0,n,8000):impulses[start:start+800]=(.004*np.exp(-np.arange(800)/90)*rng.normal(0,1,800)).astype(np.float32)
controls=[('silence',np.zeros(n)),('near_silence',rng.normal(0,1e-8,n)),('white_noise_-60dBFS',white*.001),('white_noise_-40dBFS',white*.01),('pink_noise_peak_004',pink),('hum_80Hz_peak_003',.003*np.sin(2*np.pi*80*t)),('hum_50Hz_harmonics',.001*(np.sin(2*np.pi*50*t)+np.sin(2*np.pi*100*t)+np.sin(2*np.pi*150*t))),('repeating_short_impulses',impulses),('quiet_repeating_short_impulses',impulses*.1),('synthetic_stationary_chord',.001*(np.sin(2*np.pi*220*t)+np.sin(2*np.pi*277*t)+np.sin(2*np.pi*330*t))),('fan_hum_and_noise',.001*np.sin(2*np.pi*120*t)+white*.0003)]
for label,x in controls:
 x=x.astype(np.float32)

  first,retry,selected,peak=policy(x,g);text,calls=transcribe(x,g,selected)
  item={'id':label,'mode':mode,'peak':peak,'gain':g,'defaultSpans':first,'retrySpans':retry,'selectedSpans':selected,'newFalseVad':not first and bool(selected),'hypothesis':text,'asrCalls':calls,'wouldNeedAsrModel':bool(selected),'retrySkippedByPeakGuard':not first and peak<=1e-6}
  out['controls'].append(item);save();print('negative',label,mode,'newFalseVad',item['newFalseVad'],'text',repr(text),flush=True)

save();print(json.dumps(out['summary'],indent=2),flush=True)
